2. SERVICE Y HOOKS
================================================================================
fooService.ts: las llamadas a la API.
export async function fetchFoos(firmId: number): Promise<FooOut[]> {
  const { data } = await api.get<FooOut[]>(`/firms/${firmId}/foos`);
  return data;
}
export async function createFoo(firmId: number, payload: FooCreate): Promise<FooOut> {
  const { data } = await api.post<FooOut>(`/firms/${firmId}/foos`, payload);
  return data;
}
api.get / api.post / api.patch(url, payload)
Query params: api.get(url, { params: { q } })

useFoos.ts: React Query, guarda los datos en cache.
useQuery     para leer
useMutation  para crear, editar o borrar
invalidateQueries  vuelve a pedir los datos después de un cambio

export function useFoos(firmId: number) {
  return useQuery({ queryKey: ["firm", firmId, "foos"], queryFn: () => fetchFoos(firmId) });
}
export function useCreateFoo(firmId: number) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (payload: FooCreate) => createFoo(firmId, payload),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ["firm", firmId, "foos"] }),
  });
}

En el componente:
const { data, isLoading, isError } = useFoos(firm.id);
const create = useCreateFoo(firm.id);
await create.mutateAsync(payload);      create.isPending mientras espera
